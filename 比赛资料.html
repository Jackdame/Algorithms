void merge_sort(int q[], int l, int r)
{
    //递归的终止情况
    if (l >= r) return;
    //第一步：分成子问题
    int mid = l + r >> 1;
    //第二步：递归处理子问题(可以求逆序对)
    LL res = merge_sort(q, l, mid) + merge_sort(q, mid + 1, r);
    //第三步：合并子问题
    int k = 0, i = l, j = mid + 1; 
    while (i <= mid && j <= r)
        if (q[i] <= q[j]) tmp[k ++ ] = q[i ++ ];
        else res += mid - i + 1; tmp[k ++ ] = q[j ++ ];
    while (i <= mid) tmp[k ++ ] = q[i ++ ];
    while (j <= r) tmp[k ++ ] = q[j ++ ];
    //第四步：复制回原数组
    for (i = l, j = 0; i <= r; i ++, j ++ ) q[i] = tmp[j];
}

struct DSU {
    vector<int> parent, rank_;

    DSU(int n) : parent(n), rank_(n, 0) {
        iota(parent.begin(), parent.end(), 0); // parent[i] = i
    }

    // 找根节点（路径压缩）
    int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }

    // 合并两个集合，返回是否成功（false代表本来就同一集合）
    bool unite(int a, int b) {
        a = find(a), b = find(b);
        if (a == b) return false;
        // 按秩合并：把矮树接到高树下
        if (rank_[a] < rank_[b]) swap(a, b);
        parent[b] = a;
        if (rank_[a] == rank_[b]) rank_[a]++;
        return true;
    }

    // 判断是否同一集合
    bool same(int a, int b) {
        return find(a) == find(b);
    }
};

struct BIT {
    int n;
    vector<ll> t;

    BIT(int n) : n(n), t(n + 1, 0) {}

    // 单点更新：位置 i 加上 v
    void update(int i, ll v) {
        for (; i <= n; i += i & -i) t[i] += v;
    }

    // 前缀查询：[1, i] 的和
    ll query(int i) {
        ll s = 0;
        for (; i > 0; i -= i & -i) s += t[i];
        return s;
    }

    // 区间查询：[l, r] 的和
    ll query(int l, int r) {
        return query(r) - query(l - 1);
    }
};

struct SegTree {
    int n;
    struct Node { ll sum, lazy; };
    vector<Node> t;

    SegTree(int n) : n(n), t(4 * n + 4) {}

    // ---- 内部辅助 ----
    void pushup(int o) {
        t[o].sum = t[o*2].sum + t[o*2+1].sum;
    }

    void pushdown(int o, int l, int r) {
        if (!t[o].lazy) return;
        int mid = (l + r) / 2;
        apply(o*2,   l,   mid, t[o].lazy);
        apply(o*2+1, mid+1, r, t[o].lazy);
        t[o].lazy = 0;
    }

    void apply(int o, int l, int r, ll v) {
        t[o].sum  += v * (r - l + 1);
        t[o].lazy += v;
    }

    // ---- 对外接口 ----
    // 区间加 [L,R] 每个元素 +v
    void update(int L, int R, ll v, int o = 1, int l = 1, int r = -1) {
        if (r == -1) r = n;
        if (L > r || R < l) return;
        if (L <= l && r <= R) { apply(o, l, r, v); return; }
        pushdown(o, l, r);
        int mid = (l + r) / 2;
        update(L, R, v, o*2,   l,   mid);
        update(L, R, v, o*2+1, mid+1, r);
        pushup(o);
    }

    // 区间查询 [L,R] 的和
    ll query(int L, int R, int o = 1, int l = 1, int r = -1) {
        if (r == -1) r = n;
        if (L > r || R < l) return 0;
        if (L <= l && r <= R) return t[o].sum;
        pushdown(o, l, r);
        int mid = (l + r) / 2;
        return query(L, R, o*2, l, mid) + query(L, R, o*2+1, mid+1, r);
    }
};

vector<int> slide_max(const vector<int>& a, int k){
    deque<int> dq; vector<int> res;
    req(i,0,a.size()-1)
    {
        while(!dq.empty() && a[dq.back()]<=a[i]) dq.pop_back();

struct ST {
    int n;
    vector<vector<int>> t;
    vector<int> lg;

    ST(vector<int>& a) : n(a.size()), t(20, vector<int>(a.size())), lg(a.size() + 1) {
        // 预处理 log2
        for (int i = 2; i <= n; i++) lg[i] = lg[i >> 1] + 1;
        // 初始化第一层
        t[0] = a;
        // 倍增递推
        for (int k = 1; (1 << k) <= n; k++)
            for (int i = 0; i + (1 << k) - 1 < n; i++)
                t[k][i] = min(t[k-1][i], t[k-1][i + (1 << (k-1))]);
    }

    // 查询 [l, r] 区间最小值（下标从 0 开始）
    int query(int l, int r) {
        int k = lg[r - l + 1];
        return min(t[k][l], t[k][r - (1 << k) + 1]);
    }
};

vector<LL> dj(int start, int n, vector<int> &parent)
{
    vector<LL> dist(n + 1, INF);
    vector<bool> vis(n + 1, false);
    parent.assign(n + 1, -1);
    priority_queue<PII, vector<PII>, greater<PII>> pq;
    dist[start] = 0;
    pq.push({0, start});
    while (!pq.empty())
    {
        int u = pq.top().second;
        LL d = pq.top().first;
        pq.pop();
        if (vis[u])
            continue;
        vis[u] = true;
        for (const Edge &e : g[u])
        {
            int v = e.to;
            LL w = e.weight;
            if (dist[u] + w < dist[v])
            {
                dist[v] = dist[u] + w;
                parent[v] = u;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}

int kruskal(int n) {
    sort(edges.begin(), edges.end());
    for (int i = 1; i <= n; i++) parent[i] = i;
    int total_weight = 0, cnt = 0;
    for (const auto& e : edges) {
        int pu = find(e.u), pv = find(e.v);
        if (pu != pv) {
            parent[pu] = pv;
            total_weight += e.w;
            cnt++;
            if (cnt == n - 1) break;
        }
    }
    return total_weight;
}

vector<int> Next;
void GetNext(const string& p) {
    int len = p.size();
    Next.resize(len);
    int j = 0, k = -1;
    Next[0] = -1;  
    while (j < len - 1) {
        if (k == -1 || p[k] == p[j]) {
            j++;
            k++;
            Next[j] = k;  
        }
        else {
            k = Next[k];  
        }
    }
}
int KMP(const string& t, const string& p) {
    int n = t.size(), m = p.size();
    if (m == 0) return 0;  
    GetNext(p);           
    int i = 0, j = 0;     
    while (i < n && j < m) {
        if (j == -1 || t[i] == p[j]) {  
            i++;
            j++;
        }
        else {
            j = Next[j];  
        }
    }
    if (j == m) {
        return i - j;    
    }
    else {
        return -1;        
    }
}

//简易版
struct Tree // 字典树
{
    int fail;    // 失配指针
    int vis[26]; // 子节点的位置
    int end;     // 标记有几个单词以这个节点结尾
} AC[1000000];   // Trie树
int cnt = 0;     // Trie的指针
inline void Build(string s)
{
    int l = s.length();
    int now = 0;                // 字典树的当前指针
    for (int i = 0; i < l; ++i) // 构造Trie树
    {
        if (AC[now].vis[s[i] - 'a'] == 0)    // Trie树没有这个子节点
            AC[now].vis[s[i] - 'a'] = ++cnt; // 构造出来
        now = AC[now].vis[s[i] - 'a'];       // 向下构造
    }
    AC[now].end += 1; // 标记单词结尾
}
void Get_fail() // 构造fail指针
{
    queue<int> Q;                // 队列
    for (int i = 0; i < 26; ++i) // 第二层的fail指针提前处理一下
    {
        if (AC[0].vis[i] != 0)
        {
            AC[AC[0].vis[i]].fail = 0; // 指向根节点
            Q.push(AC[0].vis[i]);      // 压入队列
        }
    }
    while (!Q.empty()) // BFS求fail指针
    {
        int u = Q.front();
        Q.pop();
        for (int i = 0; i < 26; ++i) // 枚举所有子节点
        {
            if (AC[u].vis[i] != 0) // 存在这个子节点
            {
                AC[AC[u].vis[i]].fail = AC[AC[u].fail].vis[i];
                // 子节点的fail指针指向当前节点的
                // fail指针所指向的节点的相同子节点
                Q.push(AC[u].vis[i]); // 压入队列
            }
            else // 不存在这个子节点
                AC[u].vis[i] = AC[AC[u].fail].vis[i];
            // 当前节点的这个子节点指向当
            // 前节点fail指针的这个子节点
        }
    }
}
int AC_Query(string s) // AC自动机匹配
{
    int l = s.length();
    int now = 0, ans = 0;
    for (int i = 0; i < l; ++i)
    {
        now = AC[now].vis[s[i] - 'a'];                          // 向下一层
        for (int t = now; t && AC[t].end != -1; t = AC[t].fail) // 循环求解
        {
            ans += AC[t].end;
            AC[t].end = -1;
        }
    }
    return ans;
}

打表

打表的标准流程（步骤化）

  1. 明确变量和范围：先选一个能穷尽小规模行为的范围（比如 n≤20、参数 m≤50）。
  2. 写暴力/穷举解或动态规划作为“金标准”（brute）。保证小规模下正确性。
  3. 生成表格：至少包含输入 n、输出 ans，以及可能的中间量（分解方式、余数、二进制、因子、差分、gcd 等）。
  4. 观察并尝试多种视角：差分序列、比值、因数分解、二进制位、模 p 余数、最优解结构（构造法）等。
  5. 基于表格猜想规律（形态：线性/多项式/指数/周期/分块/贪心），尝试证明（交换论证/极值法/归纳/不等式/反证）。
  6. 用对拍扩大测试：把猜想的 O(1) 算法与暴力在更大随机测试上对比，找反例并回到第 3 步修正猜想。

表格该怎么看（常用“变换”）

  1. 差分：di = ai - ai-1​，反复做差分可判定多项式程度（若 kk 次差分为常数，则原序列为 kk 次多项式）。
  2. 比值：ai/ai-1​，看是否趋近常数（指数型）。
  3. 余数/周期：看 ai mod m 是否周期性。
  4. 二进制/lowbit：检查 ai​ 的二进制模式（是否与位运算相关）。
  5. 因式分解 / 素因子分布：观察是否总有大因子或多次出现某素数。
  6. 贪心构造跟交换法：若猜到最优由若干基本块构成，尝试用交换证明“把两个块换成其他组合是否更优”。

一些思考

一个在解题中我认为最重要的一步，就是看穿题目本质，决定好方向，能想到要用什么算法。要怎么样才能做到这一步呢，我认为有几点很重要：

时间复杂度

首先，一般而言题目有两种对于T的出题方式：第一种∑n<某个上界，此时平均复杂度不会变动太大，因此主要关注∑n即可。第二种For each test case，n<某个上界，此时最坏情况就是T*O(n),因此必须考虑T的大小。对于不同n的规模，选取不同的算法：

  1. $n<12$ -> $O(n!)$ -> 全排列 dfs所有路径
  2. $n<20$ -> $O(2^n)$ -> 状压DP 简单搜索
  3. $n<40$ -> $O(2^(n/2))$ -> 折半搜索
  4. $n<100$ -> $O(n^4)$ -> Floyd最短路
  5. $n<500$ -> $O(n^3)$ -> 区间DP 高斯消元 网络流
  6. $n<5000$ -> $O(n^2)$ -> 线性二维DP Prim 直接暴力
  7. $n<210^5/510^5$ -> $O(nlogn)$ -> 排序 二分 分治 平衡树 线段树/树状数组
  8. $n<10^6$ -> $O(n)/O(nlogn)$ -> 线性筛 单调队列/栈 并查集 贪心
  9. $n<10^7$ -> 严格$O(n)$ 并使用快读和\n
  10. $n<10^5 $并且有查询 -> $O(n\sqrt{n})$ -> 莫队/分块

还有就是题目的关键词暗示

最值问题：

  1. 最大值最小/最小值最大 单调性 -> 二分答案
  2. 恰好K个/至少K个/选或不选 -> DP
  3. 按字典序最小 -> 贪心/逐位搜索
  4. 最大收益/最小花费 且有依赖关系 -> 网络流(最大流/最小割)
  5. 最小操作次数使元素相等 -> 中位数性质/找规律

方案数：

  1. 结果对mod取模 -> 组合数学/快速幂
  2. 所有下标对i,j满足i+j=k -> 卷积(FFT/NTT)
  3. 统计满足条件的区间个数 -> 双指针/树状数组
  4. 求所有可能结果之和 -> 线性性质(期望/贡献法)
  5. 计算大公约数结果 -> 莫比乌斯反演

图论：

  1. 任意两点只有唯一路径 -> 树形结构（树形DP、LCA、点分治）
  2. “没有环” 且有先后顺序依赖 -> 拓扑排序
  3. 分成了 A, B 两个集合 -> 二分图 (匹配/染色)
  4. 删掉一条边后图不连通 -> Tarjan 算法 (桥/割点)
  5. 每条边最多走一次 -> 欧拉回路 / 网络流

区间：

  1. “区间求和 / 区间最大值” + 修改 -> 线段树 / 树状数组
  2. 静态区间（不修改）询问次数巨大 -> ST表 ($O(1)$查询) 或 离线后用莫队
  3. 求不相交区间的最大覆盖 -> 贪心 (按右端点排序)
  4. 强制在线 -> 不能直接莫队或离线处理，考虑可持久化线段树 (主席树)

字符串：

  1. 寻找循环节 -> KMP 算法 的 next 数组性质
  2. 多个模式串在一个文本中寻找 -> AC 自动机
  3. 求所有子串的某种性质 -> 后缀自动机 (SAM) 或 后缀数组 (SA)

训练判题能力方法：

找到一道题屏蔽tag，先想时间复杂度，再找关键词，最后再思考结构。

常见错误

  1. 整数溢出：中间结果可能溢出，及时取模或开long long
  2. 浮点数精度：避免直接比较，使用eps
  3. 递归爆栈：DFS深度大时要改非递归或设置栈大小
  4. 多组数据：清空全局变量和容器
  5. 下标从0/1开始：统一标准
